Step 3: Create deployment.yaml
File name: deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: myapp-deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      app: myapp
  template:
    metadata:
      labels:
        app: myapp
    spec:
      containers:
        - name: myapp
          image: myapp:v1
          imagePullPolicy: IfNotPresent
          ports:
            - containerPort: 80
Step 4: Apply deployment
kubectl apply -f deployment.yaml
kubectl get deployments
kubectl get pods

Q6(A) Perform a JavaScript test using Selenium

Step 2: Create test.js
const { Builder } = require('selenium-webdriver');

async function test() {

    let driver = await new Builder()
        .forBrowser('chrome')
        .build();

    try {

        await driver.get('https://www.google.com');

        let title = await driver.getTitle();

        console.log("Page Title:", title);

        if (title.includes("Google")) {
            console.log("Test Passed");
        } else {
            console.log("Test Failed");
        }

    } finally {
        await driver.quit();
    }
}

test();
Step 3: Run the program
node test.js

const { Builder, By } = require('selenium-webdriver');

async function test() {

    let driver = await new Builder()
        .forBrowser('chrome')
        .build();

    try {

        await driver.get('http://localhost:8000/button.html');

        let button = await driver.findElement(By.id('submit'));

        await button.click();

        console.log("Button clicked successfully");

    } finally {
        await driver.quit();
    }
}

test();

import time
from selenium import webdriver
from selenium.webdriver.common.by import By

driver = None

try:

    for attempt in range(15):
        try:
            driver = webdriver.Remote(
                command_executor="http://selenium:4444",
                options=webdriver.ChromeOptions()
            )
            break
        except Exception:
            time.sleep(2)

    if driver is None:
        raise Exception("Selenium server is not ready")

    driver.get("http://webapp")

    print("Page Title:", driver.title)

    driver.find_element(By.ID, "submit").click()

    message = driver.find_element(By.ID, "message").text

    print("Message:", message)

    assert message == "Test Successful"

    print("Test Passed")

finally:
    if driver:
        driver.quit()
